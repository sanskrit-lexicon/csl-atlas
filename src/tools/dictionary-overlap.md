# ${t("phase2.overlap.title")}
How much vocabulary each pair of dictionaries shares, by normalized headword. Broad 40 is the default; Core 7 remains available as a legacy comparison scope.

## Trust Block

- Evidence: scoped broad/core payloads in `src/data/dicts/pairwise-overlap.json`.
- Limitations: overlap measures shared headwords only; it does not prove direct copying, shared senses, source dependence, or equal deep-analysis support.
- Validation: generated by `npm run build-dict-comparison`; checked by `npm run validate-dict-comparison` and `npm run build`.
- Owner repo: `csl-atlas`.
- Next use: inspect highlighted rows, then open exact dictionary source records before citing the pattern.

```js
const data = FileAttachment("../data/dicts/pairwise-overlap.json").json();
```

```js
const lang = view(Inputs.select(["en", "ru"], {
  label: "Language",
  value: "en",
  format: value => value === "ru" ? "Russian" : "English"
}));
const languageValue = String(lang ?? "").toLowerCase();
const currentLanguage = lang === 1 || languageValue === "1" || languageValue === "ru" || languageValue === "russian" || languageValue === "русский" ? "ru" : "en";
const t = (key) => {
  const currentLocale = currentLanguage === "ru" ? localesRu : localesEn;
  const parts = key.split(".");
  let result = currentLocale;
  for (const part of parts) { if (result && result[part] !== undefined) result = result[part]; else return key; }
  return result;
};
```

```js
function overlapPanel(scope, panelClass) {
  const activeData = data.scopes?.[scope] ?? data.scopes?.broadHeadword ?? data;
  const activeScopeLabel = data.scopeLabels?.[scope] ?? activeData.scopeLabel ?? scope;
  const dicts = activeData.dictionaries.map(d => d.label);
  const cells = [];
  for (const p of activeData.pairwise) {
    cells.push({ a: p.a, b: p.b, shared: p.shared, jaccard: p.jaccard });
    cells.push({ a: p.b, b: p.a, shared: p.shared, jaccard: p.jaccard });
  }
  const showCellText = scope === "coreComparison";
  const rows = activeData.pairwise.map(p => ({
    [t("phase2.overlap.dict-a")]: p.a,
    [t("phase2.overlap.dict-b")]: p.b,
    [t("phase2.overlap.shared-lemmas")]: p.shared,
    [t("phase2.overlap.jaccard")]: p.jaccard
  }));
  return html`<section class=${`scope-panel ${panelClass}`}>
    <div class="note">${t("phase2.overlap.scope-note")} <b>${activeScopeLabel}</b>.</div>
    ${Plot.plot({
      marginLeft: scope === "coreComparison" ? 60 : 90,
      marginBottom: scope === "coreComparison" ? 50 : 90,
      width: scope === "coreComparison" ? 560 : 900,
      height: scope === "coreComparison" ? 480 : 900,
      color: { scheme: "blues", label: "Jaccard", legend: true },
      x: { domain: dicts, label: null },
      y: { domain: dicts, label: null },
      marks: [
        Plot.cell(cells, { x: "a", y: "b", fill: "jaccard" }),
        ...(showCellText ? [Plot.text(cells, { x: "a", y: "b", text: d => d.jaccard.toFixed(2), fill: d => d.jaccard > 0.4 ? "white" : "black", fontSize: 11 })] : [])
      ]
    })}
    <h2>${t("phase2.overlap.pairwise-table")}</h2>
    ${Inputs.table(rows, {
      columns: [t("phase2.overlap.dict-a"), t("phase2.overlap.dict-b"), t("phase2.overlap.shared-lemmas"), t("phase2.overlap.jaccard")],
      sort: t("phase2.overlap.shared-lemmas"),
      reverse: true
    })}
  </section>`;
}

display(html`<div class="scope-switch">
```
