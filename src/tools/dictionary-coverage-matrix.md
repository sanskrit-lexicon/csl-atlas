# ${t("phase2.coverage-matrix.title")}
Headword coverage across eligible local Sanskrit/BHS dictionaries, with the legacy Core 7 comparison preserved as a selectable scope. Lemmas are grouped by normalized dictionary headword.

## Trust Block

- Evidence: scoped broad/core payloads in `src/data/dicts/coverage-matrix.json`, `src/data/dicts/dictionary-unique.json`, and `src/data/dicts/all-intersection.json`.
- Limitations: normalized lemma coverage is not sense coverage, corpus frequency, dictionary quality, or deep comparative analysis.
- Validation: generated by `npm run build-dict-comparison`; checked by `npm run validate-dict-comparison` and `npm run build`.
- Owner repo: `csl-atlas`.
- Next use: inspect highlighted rows, then open exact dictionary source records before citing the pattern.

```js
const cov = FileAttachment("../data/dicts/coverage-matrix.json").json();
const unique = FileAttachment("../data/dicts/dictionary-unique.json").json();
const intersection = FileAttachment("../data/dicts/all-intersection.json").json();
const localesEn = FileAttachment("../locales-en.json").json();
const localesRu = FileAttachment("../locales-ru.json").json();
```

```js
const lang = view(Inputs.select(["en", "ru"], {
  label: "Language",
  value: "en",
  format: value => value === "ru" ? "Russian" : "English"
}));
const languageValue = String(lang ?? "").toLowerCase();
const currentLanguage = lang === 1 || languageValue === "1" || languageValue === "ru" || languageValue === "russian" || languageValue === "русский" ? "ru" : "en";
const t = (key) => {
  const currentLocale = currentLanguage === "ru" ? localesRu : localesEn;
  const parts = key.split(".");
  let result = currentLocale;
  for (const part of parts) { if (result && result[part] !== undefined) result = result[part]; else return key; }
  return result;
};
```

```js
function coveragePanel(scope, panelClass) {
  const activeCov = cov.scopes?.[scope] ?? cov.scopes?.broadHeadword ?? cov;
  const activeUnique = unique.scopes?.[scope] ?? unique.scopes?.broadHeadword ?? unique;
  const activeIntersection = intersection.scopes?.[scope] ?? intersection.scopes?.broadHeadword ?? intersection;
  const activeScopeLabel = cov.scopeLabels?.[scope] ?? activeCov.scopeLabel ?? scope;
  const byDict = Object.entries(activeCov.lemmasByDict).map(([dict, lemmas]) => ({ dict, lemmas, unique: activeUnique.unique[dict]?.count ?? 0 }));
  const hist = Object.entries(activeCov.coverageHistogram).map(([n, count]) => ({ n: +n, count })).sort((a, b) => a.n - b.n);
  const comboRows = activeCov.topCombinations.map(c => ({
    [t("phase2.coverage-matrix.combination")]: c.dicts.join(" + "),
    [t("phase2.coverage-matrix.dictionaries")]: c.size,
    [t("phase2.coverage-matrix.lemmas")]: c.count
  }));
  return html`<section class=${`scope-panel ${panelClass}`}>
    <div class="note">${t("phase2.coverage-matrix.scope-note")} <b>${activeScopeLabel}</b>.</div>
    <div style="display:flex;gap:24px;flex-wrap:wrap;margin:8px 0">
      ${[
        [t("phase2.coverage-matrix.distinct-lemmas"), activeCov.distinctLemmas.toLocaleString()],
        [t("phase2.coverage-matrix.shared-by-all"), activeIntersection.count.toLocaleString()],
        [t("phase2.coverage-matrix.dictionaries"), Object.keys(activeCov.lemmasByDict).length]
      ].map(([l, v]) => html`<div><div style="font-size:1.8rem;font-weight:700">${v}</div><div style="color:var(--theme-foreground-muted);font-size:.85rem">${l}</div></div>`)}
    </div>
    <h2>${t("phase2.coverage-matrix.lemmas-per-dict")}</h2>
    ${Plot.plot({
      marginLeft: 60,
      height: Math.max(320, byDict.length * 18),
      x: { label: t("phase2.coverage-matrix.distinct-lemmas"), grid: true },
      y: { label: null, sort: { y: "x", reverse: true } },
      marks: [
        Plot.barX(byDict, { x: "lemmas", y: "dict", fill: "var(--theme-foreground-faint)" }),
        Plot.barX(byDict, { x: "unique", y: "dict", fill: "var(--theme-foreground-focus)" }),
        Plot.ruleX([0])
      ]
    })}
    <div class="note">Dark bar = lemmas unique to that dictionary within the selected scope.</div>
    <h2>${t("phase2.coverage-matrix.how-many-cover")}</h2>
    ${Plot.plot({
      height: 280,
      x: { label: t("phase2.coverage-matrix.in-how-many"), tickFormat: "d" },
      y: { label: t("phase2.coverage-matrix.lemmas"), grid: true },
      marks: [Plot.barY(hist, { x: "n", y: "count", fill: "var(--theme-foreground-focus)" }), Plot.ruleY([0])]
    })}
    <h2>${t("phase2.coverage-matrix.most-common-combinations")}</h2>
    ${Inputs.table(comboRows, {
      columns: [t("phase2.coverage-matrix.combination"), t("phase2.coverage-matrix.dictionaries"), t("phase2.coverage-matrix.lemmas")],
      sort: "lemmas",
      reverse: true
    })}
  </section>`;
}

display(html`<div class="scope-switch">

  <input id="coverage-scope-core" type="radio" name="coverage-scope">
  <label for="coverage-scope-core">${cov.scopeLabels?.coreComparison ?? "Core 7"}</label>
  ${coveragePanel("broadHeadword", "scope-panel-broad")}
  ${coveragePanel("coreComparison", "scope-panel-core")}
  <style>

</div>`);
```
